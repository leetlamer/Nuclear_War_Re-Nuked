cmake_minimum_required(VERSION 3.20)
project(NuclearWarReNuked VERSION 1.0.0 LANGUAGES CXX)
include(CTest)
find_package(SDL2 CONFIG REQUIRED)
find_package(PNG REQUIRED)
option(NWR_ENABLE_UBSAN "Enable undefined-behavior checks (Clang/GCC)" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NWR_ENABLE_UBSAN AND NOT MSVC)
 add_compile_options(-fsanitize=undefined -fno-sanitize-recover=all)
 add_link_options(-fsanitize=undefined -fno-sanitize-recover=all)
endif()
add_library(nuclear_game
 src/game/game_rules.cpp src/game/scoring.cpp src/game/ai/round_plan.cpp
 src/game/combat/bomber_navigation.cpp src/game/combat/resolver_markers.cpp
 src/game/combat/turn_resolver.cpp src/game/combat/missile_attack.cpp
 src/game/turn_manager.cpp src/ui/human_commands.cpp src/ui/action_messages.cpp
 src/core/game_text.cpp src/ui/player_controller.cpp)
target_include_directories(nuclear_game PUBLIC include src "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(nuclear_game PUBLIC cxx_std_17)
add_executable(nuclearwar src/platform/application.cpp)
target_link_libraries(nuclearwar PRIVATE nuclear_game SDL2::SDL2 PNG::PNG)
if(TARGET SDL2::SDL2main)
 target_link_libraries(nuclearwar PRIVATE SDL2::SDL2main)
endif()
if(BUILD_TESTING)
 function(game_test name source)
  add_executable(${name} ${source} ${ARGN})
  target_include_directories(${name} PRIVATE tests/fixtures)
  target_link_libraries(${name} PRIVATE nuclear_game SDL2::SDL2 PNG::PNG)
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/.migration/baseline" ENVIRONMENT "SDL_VIDEODRIVER=dummy")
 endfunction()
 foreach(name engine_tests ending_tests resolver_marker_tests round_plan_tests)
  game_test(${name} tests/engine/${name}.cpp)
 endforeach()
 game_test(session_tests tests/session/session_tests.cpp)
 game_test(resolver_tests tests/resolver/resolver_core_tests.cpp)
 game_test(missile_tests tests/resolver/single_missile_tests.cpp)
 game_test(text_tests tests/core/game_text_tests.cpp)
 game_test(player_tests tests/frontend/tests.cpp)
 game_test(revenge_tests tests/frontend/revenge_tests.cpp)
 game_test(audio_tests tests/frontend/audio_tests.cpp)
 game_test(render_tests tests/frontend/render_tests.cpp)
 game_test(legacy_prototype_tests tests/support/legacy_prototype/tests.cpp tests/support/legacy_prototype/game.cpp)
 game_test(missile_demo_tests tests/support/missile_demo/tests.cpp tests/support/missile_demo/model.cpp)
endif()
